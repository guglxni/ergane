5. Codex CLI Setup

5.1 Official Install (Recommended)

5.2 Configure Codex Agents

# Hermes creates agent configs
mkdir -p ~/ergane/.codex/agents

# Copy from SKILL.html specs (already in docs/SKILL.html)
# Then configure Codex to use local agents:
codex config set agents.dir ~/ergane/.codex/agents

# Authenticate with OpenAI
codex auth

# Verify model access (make sure gpt-5.6 is available in your plan)
codex models list | grep gpt-5.6

5.3 Run Codex on Ergane

# Hermes defines task; Codex executes
cd ~/ergane
codex exec --approval-mode full-auto \
  --agent ergane-impl \
  "Implement the CLI scan command per docs/ARCHITECTURE.html Stage 1-2. Use gpt-5.6-terra for detection."

# Generate /feedback Session ID (needed for hackathon submission)
codex exec "Generate a /feedback session summary"

6. Coral Setup

6.1 Install Coral CLI

# Hermes installs
brew install withcoral/tap/coral

# Verify
coral --version  # Expected: >= 0.5.2

# Add MCP server
mkdir -p ~/.config/coral
coral mcp-stdio  # Run MCP stdio server for agent integration

6.2 Configure Patent Sources

# Hermes writes source specs (from docs/ARCHITECTURE.html)
mkdir -p ~/.config/coral/sources

# Copy corrected specs
cp ~/ergane/coral_sources/pqai.yaml ~/.config/coral/sources/
cp ~/ergane/coral_sources/uspto.yaml ~/.config/coral/sources/
cp ~/ergane/coral_sources/epo.yaml ~/.config/coral/sources/

# Add sources to Coral
coral source add pqai ~/.config/coral/sources/pqai.yaml
coral source add uspto ~/.config/coral/sources/uspto.yaml
coral source add epo ~/.config/coral/sources/epo.yaml

# List and validate
coral source list
coral source validate pqai
coral source validate uspto
